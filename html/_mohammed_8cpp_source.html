17template <typename BoardType>
18Move<char>* GameAI::getBestMove_2D(Player<char>* player) {
19 BoardType* board = dynamic_cast<BoardType*>(player->get_board_ptr());
20 if (!board) return nullptr;
21
22 char aiSym = player->get_symbol();
23 char humanSym = (aiSym == 'X') ? 'O' : 'X';
24 int rows = board->get_rows();
25 int cols = board->get_columns();
26
27 // Strategy 1: Check for an immediate win
28 for (int r = 0; r < rows; r++) {
29 for (int c = 0; c < cols; c++) {
30 Move<char>* testMove = new Move<char>(r, c, aiSym);
31 if (board->update_board(testMove)) {
32 if (board->is_win(player)) {
33 board->undoLastMove();
34 return new Move<char>(r, c, aiSym);
35 }
36 board->undoLastMove();
37 }
38 }
39 }
40
41 // Strategy 2: Check for an immediate block (prevent opponent win)
42 Player<char> tempHuman("temp", humanSym, PlayerType::HUMAN);
43 for (int r = 0; r < rows; r++) {
44 for (int c = 0; c < cols; c++) {
45 Move<char>* testMove = new Move<char>(r, c, humanSym);
46 if (board->update_board(testMove)) {
47 if (board->is_win(&tempHuman)) {
48 board->undoLastMove();
49 return new Move<char>(r, c, aiSym);
50 }
51 board->undoLastMove();
52 }
53 }
54 }
55
56 // Strategy 3: Random move fallback
57 int r, c;
58 Move<char>* randomMove;
59 do {
60 r = rand() % rows;
61 c = rand() % cols;
62 randomMove = new Move<char>(r, c, aiSym);
63 } while (!board->update_board(randomMove));
64
65 board->undoLastMove();
66 return new Move<char>(r, c, aiSym);
67}

115template <typename BoardType>
116Move<char>* GameAI::getMisereMove(Player<char>* player) {
117 BoardType* board = dynamic_cast<BoardType*>(player->get_board_ptr());
118 if (!board) return nullptr;
119
120 char aiSym = player->get_symbol();
121 int rows = board->get_rows();
122 int cols = board->get_columns();
123
124 vector<Point> safeMoves;
125 vector<Point> allValidMoves;
126
127 for (int r = 0; r < rows; r++) {
128 for (int c = 0; c < cols; c++) {
129 Move<char>* testMove = new Move<char>(r, c, aiSym);
130 if (board->update_board(testMove)) {
131 allValidMoves.push_back({r, c});
132 // In Misere, is_lose checks if 3-in-a-row is formed
133 if (!board->is_lose(player)) {
134 safeMoves.push_back({r, c});
135 }
136 board->undoLastMove();
137 }
138 }
139 }
140
141 Point p;
142 if (!safeMoves.empty()) {
143 p = safeMoves[rand() % safeMoves.size()];
144 } else if (!allValidMoves.empty()) {
145 p = allValidMoves[rand() % allValidMoves.size()];
146 } else {
147 p = {0, 0};
148 }
149 return new Move<char>(p.r, p.c, aiSym);
150}

268bool C4_Board::checkWin(char s) {
269 for (int r = 0; r < rows; r++) {
270 for (int c = 0; c < columns; c++) {
271 if (board[r][c] == s) {
272 // Check Horizontal, Vertical, Diagonal
273 if (c + 3 < columns && board[r][c + 1] == s && board[r][c + 2] == s && board[r][c + 3] == s) return true;
274 if (r + 3 < rows && board[r + 1][c] == s && board[r + 2][c] == s && board[r + 3][c] == s) return true;
275 if (r + 3 < rows && c + 3 < columns && board[r + 1][c + 1] == s && board[r + 2][c + 2] == s && board[r + 3][c + 3] == s) return true;
276 if (r - 3 >= 0 && c + 3 < columns && board[r - 1][c + 1] == s && board[r - 2][c + 2] == s && board[r - 3][c + 3] == s) return true;
277 }
278 }
279 }
280 return false;

848
849 } while (!board->public_is_valid(r, c, num, is_odd));

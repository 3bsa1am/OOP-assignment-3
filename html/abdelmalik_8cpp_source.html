174bool Diamond_XO_Board::is_win(Player<char>* player) {
175 char s = player->get_symbol();
176 set<int> dirs3, dirs4;
177
178 // Directions: 0:Right, 1:Down, 2:DiagDownRight, 3:DiagDownLeft
179 int dr[] = {0, 1, 1, 1};
180 int dc[] = {1, 0, 1, -1};
181
182 auto get_len = [&](int r, int c, int d) {
183 int len = 0;
184 int tr = r, tc = c;
185 // Check bounds for 7x7
186 while(tr >= 0 && tr < 7 && tc >= 0 && tc < 7 && board[tr][tc] == s) {
187 len++;
188 tr += dr[d];
189 tc += dc[d];
190 }
191 return len;
192 };
193
194 auto is_start_of_line = [&](int r, int c, int d) {
195 int pr = r - dr[d];
196 int pc = c - dc[d];
197 // If previous cell is valid and same symbol, this isn't the start
198 if (pr >= 0 && pr < 7 && pc >= 0 && pc < 7 && board[pr][pc] == s) return false;
199 return true;
200 };
201
202 for(int i=0; i<7; i++) {
203 for(int j=0; j<7; j++) {
204 if(board[i][j] == s) {
205 for(int d=0; d<4; d++) {
206 if(is_start_of_line(i, j, d)) {
207 int len = get_len(i, j, d);
208 // A line of 4 implies a line of 3, so add to both
209 if(len >= 3) dirs3.insert(d);
210 if(len >= 4) dirs4.insert(d);
211 }
212 }
213 }
214 }
215 }
216
217 // Win if we have a line of 3 and a line of 4 in DIFFERENT directions
218 for(int d3 : dirs3) {
219 for(int d4 : dirs4) {
220 if(d3 != d4) return true;
221 }
222 }
223
224 return false;
225}

355void Memory_XO_UI::display_board_matrix(const vector<vector<char>>& matrix) const {
356 // Check if the game is over (Win or Draw) to reveal marks
357 bool reveal = false;
358
359 // Check Rows & Cols for win
360 for(int i=0; i<3; ++i) {
361 if(matrix[i][0] != 0 && matrix[i][0] == matrix[i][1] && matrix[i][1] == matrix[i][2]) reveal = true;
362 if(matrix[0][i] != 0 && matrix[0][i] == matrix[1][i] && matrix[1][i] == matrix[2][i]) reveal = true;
363 }
364 // Check Diagonals for win
365 if(matrix[0][0] != 0 && matrix[0][0] == matrix[1][1] && matrix[1][1] == matrix[2][2]) reveal = true;
366 if(matrix[0][2] != 0 && matrix[0][2] == matrix[1][1] && matrix[1][1] == matrix[2][0]) reveal = true;
367
368 // Check for Full Board (Draw)
369 if (!reveal) {
370 bool full = true;
371 for(const auto& r : matrix) {
372 for(char c : r) if(c == 0) full = false;
373 }
374 if(full) reveal = true;
375 }
376
377 cout << "\n";
378 cout << " 0 1 2\n";
379 cout << " -------------\n";
380
381 for (int i = 0; i < 3; ++i) {
382 cout << " " << i << " | ";
383 for (int j = 0; j < 3; ++j) {
384 char cell = matrix[i][j];
385 char displayChar;
386
387 if (cell == 0) {
388 displayChar = '.'; // Empty cell
389 } else if (reveal) {
390 displayChar = cell; // Game Over: Show actual mark
391 } else {
392 displayChar = '#'; // Game Active: Show mask
393 }
394
395 cout << displayChar << " | ";
396 }
397 cout << "\n -------------\n";
398 }

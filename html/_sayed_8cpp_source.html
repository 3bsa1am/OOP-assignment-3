57// Counts how many "3-in-a-rows" exist for a specific symbol
58int XO_5x5_Board::count_three_in_row(char symbol) const {
59 char sym = toupper(symbol);
60 int count = 0;
61
62 // Check Rows
63 for (int i = 0; i < rows; i++) {
64 for (int j = 0; j <= columns - 3; j++) {
65 if (board[i][j] == sym && board[i][j+1] == sym && board[i][j+2] == sym) count++;
66 }
67 }
68 // Check Cols
69 for (int j = 0; j < columns; j++) {
70 for (int i = 0; i <= rows - 3; i++) {
71 if (board[i][j] == sym && board[i+1][j] == sym && board[i+2][j] == sym) count++;
72 }
73 }
74 // Check Diagonals
75 for (int i = 0; i <= rows - 3; i++) {
76 for (int j = 0; j <= columns - 3; j++) {
77 if (board[i][j] == sym && board[i+1][j+1] == sym && board[i+2][j+2] == sym) count++;
78 if (board[i][j+2] == sym && board[i+1][j+1] == sym && board[i+2][j] == sym) count++;
79 }
80 }
81 return count;
82}

114Move<char>* XO_5x5_UI::computerMove(Player<char>* aiPlayer) {
115 XO_5x5_Board* board = dynamic_cast<XO_5x5_Board*>(aiPlayer->get_board_ptr());
116 if (!board) return nullptr;
117
118 char aiSym = aiPlayer->get_symbol();
119 char humanSym = (aiSym == 'X') ? 'O' : 'X';
120 int rows = board->get_rows();
121 int cols = board->get_columns();
122
123 int currentAiScore = board->count_three_in_row(aiSym);
124 int currentHumanScore = board->count_three_in_row(humanSym);
125
126 // 1. Try to increase own score
127 for (int r = 0; r < rows; r++) {
128 for (int c = 0; c < cols; c++) {
129 Move<char>* testMove = new Move<char>(r, c, aiSym);
130 if (board->update_board(testMove)) {
131 if (board->count_three_in_row(aiSym) > currentAiScore) {
132 board->undoLastMove();
133 return new Move<char>(r, c, aiSym);
134 }
135 board->undoLastMove();
136 }
137 }
138 }
139
140 // 2. Block opponent
141 for (int r = 0; r < rows; r++) {
142 for (int c = 0; c < cols; c++) {
143 Move<char>* testMove = new Move<char>(r, c, humanSym);
144 if (board->update_board(testMove)) {
145 if (board->count_three_in_row(humanSym) > currentHumanScore) {
146 board->undoLastMove();
147 return new Move<char>(r, c, aiSym);
148 }
149 board->undoLastMove();
150 }
151 }
152 }
153
154 // 3. Random fallback
155 return GameAI::getRandomMove<XO_5x5_Board>(aiPlayer);
156}

210bool Word_TTT_Board::load_dictionary(const string& filename) {
211 ifstream file(filename);

215
216 string word;
217 while (file >> word) {
218 // Sanitize input

318Move<char>* Word_TTT_UI::computerMove(Player<char>* aiPlayer) {
319 Word_TTT_Board* board = dynamic_cast<Word_TTT_Board*>(aiPlayer->get_board_ptr());
320 if (!board) return nullptr;
321
322 int rows = 3;
323 int cols = 3;
324
325 // 1. Try A-Z to see if AI can win
326 for (int r = 0; r < rows; r++) {
327 for (int c = 0; c < cols; c++) {
328 if (board->get_cell(r, c) == ' ') {
329 for (char l = 'A'; l <= 'Z'; l++) {
330 Move<char>* testMove = new Move<char>(r, c, l);
331 if (board->update_board(testMove)) {
332 if (board->is_win(aiPlayer)) {
333 board->undoLastMove();
334 return new Move<char>(r, c, l);
335 }
336 board->undoLastMove();
337 }
338 }
339 }
340 }
341 }
342
343 // 2. Play Random valid move
344 vector<Point> emptyCells;
345 for(int r=0; r<3; r++)
346 for(int c=0; c<3; c++)
347 if(board->get_cell(r,c) == ' ') emptyCells.push_back({r,c});
348
349 if(!emptyCells.empty()){
350 Point p = emptyCells[rand() % emptyCells.size()];
351 char l = 'A' + (rand() % 26);
352 return new Move<char>(p.r, p.c, l);
353 }
354
355 return new Move<char>(0, 0, 'A');

385bool Infinity_TTT_Board::check_win(char symbol) const {
386 char sym = toupper(symbol);
387 // Rows
388 for (int i = 0; i < 3; i++) {
389 if (board[i][0] == sym && board[i][1] == sym && board[i][2] == sym) return true;
390 }
391 // Cols
392 for (int j = 0; j < 3; j++) {
393 if (board[0][j] == sym && board[1][j] == sym && board[2][j] == sym) return true;
394 }
395 // Diagonals
396 if (board[0][0] == sym && board[1][1] == sym && board[2][2] == sym) return true;
397 if (board[0][2] == sym && board[1][1] == sym && board[2][0] == sym) return true;
398

402bool Infinity_TTT_Board::update_board(Move<char>* move) {
403 int x = move->get_x();
404 int y = move->get_y();
405 char sym = move->get_symbol();
406
407 if (x >= 0 && x < rows && y >= 0 && y < columns && board[x][y] == blank_symbol) {
408 n_moves++;
409 board[x][y] = toupper(sym);
410 move_queue.push_back({x, y});
411
412 InfinityMoveInfo info;
413 info.placed = {x, y};
414 info.caused_removal = false;
415
416 // Removal Condition: Oldest move disappears every 3 moves starting from move 4
417 // Typically Infinity TTT limits pieces to 3 per player (6 total).
418 // If we strictly follow "every 3 moves", board fills up.
419 // But if user wants a draw on full board, we let it fill.
420
421 if (n_moves >= 4 && (n_moves - 1) % 3 == 0) {
422 if (!move_queue.empty()) {
423 Point oldest = move_queue.front();
424 move_queue.pop_front();
425
426 info.caused_removal = true;
427 info.removed_point = oldest;
428 info.removed_symbol = board[oldest.r][oldest.c];
429
430 board[oldest.r][oldest.c] = blank_symbol;
431 }
432 }
433
434 history.push_back(info);
435 delete move;
436 return true;
437 }
438
439 delete move;
440 return false;
441}

443void Infinity_TTT_Board::undoLastMove() {
444 if (!history.empty()) {
445 InfinityMoveInfo last = history.back();
446 history.pop_back();
447
448 // 1. Remove the placed piece
449 board[last.placed.r][last.placed.c] = blank_symbol;
450 move_queue.pop_back();
451
452 // 2. Restore removed piece
453 if (last.caused_removal) {
454 board[last.removed_point.r][last.removed_point.c] = last.removed_symbol;
455 move_queue.push_front(last.removed_point);
456 }
457
458 n_moves--;
459 }
